AI全域运营系统 KnowledgeSpace 数据结构草案 v1

本草案用于把知识空间从概念对象推进到可定义接口、可拆表、可支撑后台页面的真实实现层。

核心结论： 建议以 KnowledgeSpace 作为知识库域聚合根，统一承接文档、标签、绑定关系和默认检索配置。

1. 结构分层

  1. 主对象层：KnowledgeSpace
  2. 内容与标签层：KnowledgeDocument、KnowledgeTag
  3. 绑定与配置层：KnowledgeBinding、KnowledgeRetrievalConfig

2. 核心结构示例

export interface KnowledgeSpace {
  id: string;
  spaceKey: string;
  spaceName: string;
  scope: "PLATFORM" | "BRAND" | "MODULE" | "USER";
  ownerBrandId?: string;
  ownerUserId?: string;
  moduleKeys: string[];
  status: "DRAFT" | "ACTIVE" | "DISABLED" | "ARCHIVED";
  sourceMode: "MANUAL" | "SYNCED" | "HYBRID";
  visibility: "PRIVATE" | "BRAND_SHARED" | "PLATFORM_SHARED";
  tagIds: string[];
  retrievalConfigId?: string;
}
export interface KnowledgeDocument {
  id: string;
  spaceId: string;
  documentKey: string;
  title: string;
  sourceType: "FILE" | "URL" | "DOC" | "MARKDOWN" | "API";
  parseStatus: "PENDING" | "PARSED" | "FAILED";
  syncStatus: "NONE" | "SYNCING" | "SYNCED" | "FAILED";
}

3. 列表与详情 DTO

建议统一提供 KnowledgeSpaceListResponse 和 KnowledgeSpaceDetailResponse，让列表页、详情页和绑定摘要都建立在同一套主对象上。

4. 第一阶段最小落地

P0： KnowledgeSpace、KnowledgeDocument、KnowledgeBinding

P1： KnowledgeTag、KnowledgeRetrievalConfig
